# PHASE 8 — TG's Restaurant ERP: Automatic Weekly Google Drive Backup + Database Clear Engine
### (ቲጂ ምግብ ቤት — Replit Agent Brief)

Read every section fully before writing a single line of code. This phase adds a powerful automatic weekly backup system that exports all operational data to Google Drive as Excel files, then clears the database — keeping the Neon free tier alive forever while preserving all historical data safely in Google Drive.

---

## 0. WHAT THIS SYSTEM DOES

Every Sunday at midnight UAE time (UTC+4 = 20:00 UTC Saturday):
1. Exports all operational/transactional tables to one Excel file (multiple sheets)
2. Uploads the Excel file to a configured Google Drive folder automatically
3. Clears the exported data from the database
4. Sends a notification to the admin (WhatsApp + Teams if configured)
5. Logs the backup in a `backup_logs` table

The admin can also trigger this manually anytime from the Settings page with one click.

**Data that IS backed up and cleared weekly:**
- orders + order_items + order_status_history
- deliveries
- lottery_entries + lottery_draws + lottery_winners
- finance_entries + expenses
- commissions
- timesheets
- whatsapp_messages
- import_shipments + import_shipment_items + import_payments
- audit_logs
- login_attempts
- staff_activities (completed ones only)

**Data that is NEVER touched:**
- users (staff accounts, phones, hashed passwords, roles)
- branches
- menu_items + menu_categories
- inventory_items
- suppliers_addis
- settings
- restock_orders (pending ones stay)
- customers (kept for history — phone numbers needed for lottery)

### 1.2 Google Drive Authentication

Use Google Drive API v3 with a **Service Account** (not OAuth — service accounts work server-side without user interaction).

Create a utility file `artifacts/api-server/src/lib/google-drive.ts`:

```typescript
import { google } from 'googleapis';
import { Readable } from 'stream';

export async function getGoogleDriveClient() {
  const clientEmail = await getSetting('google_drive_client_email');
  const privateKey = await getSetting('google_drive_private_key');
  const folderId = await getSetting('google_drive_folder_id');

  if (!clientEmail || !privateKey) {
    throw new Error('Google Drive credentials not configured in Settings');
  }

  const auth = new google.auth.GoogleAuth({
    credentials: {
      client_email: clientEmail,
      private_key: privateKey.replace(/\\n/g, '\n'), // fix escaped newlines
    },
    scopes: ['https://www.googleapis.com/auth/drive.file'],
  });

  const drive = google.drive({ version: 'v3', auth });
  return { drive, folderId };
}

export async function uploadFileToDrive(
  fileName: string,
  buffer: Buffer,
  mimeType: string,
  folderId: string | null
): Promise<{ fileId: string; webViewLink: string }> {
  const { drive } = await getGoogleDriveClient();

  const fileMetadata: any = { name: fileName };
  if (folderId) fileMetadata.parents = [folderId];

  const media = {
    mimeType,
    body: Readable.from(buffer),
  };

  const response = await drive.files.create({
    requestBody: fileMetadata,
    media,
    fields: 'id, webViewLink',
  });

  return {
    fileId: response.data.id!,
    webViewLink: response.data.webViewLink!,
  };
}
```

## 3. BACKUP + CLEAR ENGINE

Create `artifacts/api-server/src/lib/weekly-backup.ts`:

```typescript
import { generateBackupExcel } from './backup-generator';
import { uploadFileToDrive } from './google-drive';
import { getSetting } from './settings';
import { sendTeamsNotification } from './teams';
import { db } from '@workspace/db';
import {
  orders, orderItems, orderStatusHistory,
  deliveries, lotteryEntries, lotteryDraws, lotteryWinners,
  financeEntries, expenses, commissions, timesheets,
  whatsappMessages, importShipments, importShipmentItems,
  importPayments, auditLogs, loginAttempts, backupLogs
} from '@workspace/db/schema';

export async function runWeeklyBackup(): Promise<{
  success: boolean;
  fileId?: string;
  webViewLink?: string;
  error?: string;
  rowsCleared: number;
}> {
  const now = new Date();
  const weekLabel = now.toISOString().split('T')[0]; // e.g. "2026-07-11"
  const fileName = `TG-Restaurant-Backup-Week-${weekLabel}.xlsx`;

  let rowsCleared = 0;

  try {
    // Step 1 — Generate Excel
    console.log('[Backup] Generating Excel file...');
    const buffer = await generateBackupExcel(weekLabel);
    console.log(`[Backup] Excel generated — ${(buffer.length / 1024).toFixed(1)} KB`);

    // Step 2 — Upload to Google Drive
    console.log('[Backup] Uploading to Google Drive...');
    const folderId = await getSetting('google_drive_folder_id');
    const { fileId, webViewLink } = await uploadFileToDrive(
      fileName,
      buffer,
      'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
      folderId
    );
    console.log(`[Backup] Uploaded — file ID: ${fileId}`);

    // Step 3 — Clear operational data (in correct FK order)
    console.log('[Backup] Clearing operational data...');

    // Delete in FK-safe order (children before parents)
    await db.delete(loginAttempts);
    await db.delete(auditLogs);
    await db.delete(lotteryWinners);
    await db.delete(lotteryEntries);
    await db.delete(lotteryDraws);
    await db.delete(commissions);
    await db.delete(timesheets);
    await db.delete(financeEntries);
    await db.delete(expenses);
    await db.delete(whatsappMessages);
    await db.delete(importPayments);
    await db.delete(importShipmentItems);
    await db.delete(importShipments);
    await db.delete(orderStatusHistory);
    await db.delete(orderItems);
    await db.delete(deliveries);
    const deletedOrders = await db.delete(orders).returning();
    rowsCleared = deletedOrders.length;

    console.log(`[Backup] Cleared ${rowsCleared} orders and all related data`);

    // Step 4 — Log the backup
    await db.insert(backupLogs).values({
      weekLabel,
      fileName,
      fileId,
      webViewLink,
      rowsCleared,
      status: 'success',
      createdAt: now,
    });

    // Step 5 — Send notifications
    const teamsWebhook = await getSetting('microsoft_teams_webhook_url');
    if (teamsWebhook) {
      await sendTeamsNotification(
        teamsWebhook,
        '✅ Weekly Backup Complete',
        `TG's Restaurant data for week of ${weekLabel} has been backed up to Google Drive and the database has been cleared.\n\n📊 Orders backed up: ${rowsCleared}\n📁 File: ${fileName}\n🔗 [View in Google Drive](${webViewLink})`,
        '00B050'
      );
    }

    console.log('[Backup] Weekly backup complete ✅');
    return { success: true, fileId, webViewLink, rowsCleared };

  } catch (error: any) {
    console.error('[Backup] Failed:', error);

    // Log the failure
    await db.insert(backupLogs).values({
      weekLabel,
      fileName,
      fileId: null,
      webViewLink: null,
      rowsCleared: 0,
      status: 'failed',
      errorMessage: error.message,
      createdAt: now,
    }).catch(() => {}); // Don't throw if log insert fails

    return { success: false, error: error.message, rowsCleared: 0 };
  }
}
```

## 6. API ROUTES — BACKUP MANAGEMENT

Add a new route file `artifacts/api-server/src/routes/backup.ts`:

```typescript
import { Router } from 'express';
import { authenticate, requireRole } from '../middlewares/auth';
import { runWeeklyBackup } from '../lib/weekly-backup';
import { db } from '@workspace/db';
import { backupLogs } from '@workspace/db/schema';
import { desc } from 'drizzle-orm';

const router = Router();
router.use(authenticate, requireRole('super_admin'));

// GET /api/backup/logs — last 20 backup history entries
router.get('/logs', async (req, res) => {
  const logs = await db.select().from(backupLogs)
    .orderBy(desc(backupLogs.createdAt))
    .limit(20);
  res.json(logs);
});

// POST /api/backup/run — manually trigger backup now
router.post('/run', async (req, res) => {
  // Run in background — don't await (could take 30-60 seconds)
  res.json({ message: 'Backup started. Check backup logs in a few minutes.' });
  
  runWeeklyBackup().then(result => {
    console.log('[Manual Backup]', result);
  }).catch(err => {
    console.error('[Manual Backup] Error:', err);
  });
});

// GET /api/backup/test-drive — test Google Drive connection
router.get('/test-drive', async (req, res) => {
  try {
    const { getGoogleDriveClient } = await import('../lib/google-drive');
    const { drive } = await getGoogleDriveClient();
    const about = await drive.about.get({ fields: 'user' });
    res.json({ 
      success: true, 
      connectedAs: about.data.user?.emailAddress,
      message: 'Google Drive connection successful'
    });
  } catch (err: any) {
    res.status(400).json({ success: false, error: err.message });
  }
});

export default router;
```

Register in `artifacts/api-server/src/routes/index.ts`:
